318 class E57Exception : public std::exception
319 {
320 public:
321 const char *what() const noexcept override
322 {
323 return "E57 exception";
324 }
325
326 void report( const char *reportingFileName = nullptr, int reportingLineNumber = 0,
327 const char *reportingFunctionName = nullptr,
328 std::ostream &os = std::cout ) const noexcept
329 {
330 os << "**** Got an e57 exception: " << errorStr() << std::endl;
331
332#ifdef E57_ENABLE_DIAGNOSTIC_OUTPUT
333 os << " Debug info: " << std::endl;
334 os << " context: " << context_ << std::endl;
335 os << " sourceFunctionName: " << sourceFunctionName_ << std::endl;
336 if ( reportingFunctionName != nullptr )
337 {
338 os << " reportingFunctionName: " << reportingFunctionName << std::endl;
339 }
340
341 /*** Add a line in error message that a smart editor (gnu emacs) can
342 * interpret as a link to the source code: */
343 os << sourceFileName_ << "(" << sourceLineNumber_ << ") : error C" << errorCode_
344 << ": <--- occurred on" << std::endl;
345 if ( reportingFileName != nullptr )
346 {
347 os << reportingFileName << "(" << reportingLineNumber
348 << ") : error C0: <--- reported on" << std::endl;
349 }
350#else
351 E57_UNUSED( reportingFileName );
352 E57_UNUSED( reportingLineNumber );
353 E57_UNUSED( reportingFunctionName );
354#endif
355 }
356
357 ErrorCode errorCode() const noexcept
358 {
359 return errorCode_;
360 }
361
362 std::string errorStr() const noexcept
363 {
364 return Utilities::errorCodeToString( errorCode_ );
365 }
366
367 const std::string &context() const noexcept
368 {
369 return context_;
370 }
371
372 // For debugging purposes:
373 const char *sourceFileName() const noexcept
374 {
375 return sourceFileName_.c_str();
376 }
377
378 const char *sourceFunctionName() const noexcept
379 {
380 return sourceFunctionName_;
381 }
382
383 int sourceLineNumber() const noexcept
384 {
385 return sourceLineNumber_;
386 }
387
389 E57Exception() = delete;
390 E57Exception( ErrorCode ecode, std::string context, const char *srcFileName = nullptr,
391 int srcLineNumber = 0, const char *srcFunctionName = nullptr ) :
392 errorCode_( ecode ), context_( std::move( context ) ), sourceFileName_( srcFileName ),
393 sourceFunctionName_( srcFunctionName ), sourceLineNumber_( srcLineNumber )
394 {
395 }

398 private:
400 ErrorCode errorCode_;
